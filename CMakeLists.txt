cmake_minimum_required(VERSION 3.20)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED True)
project(tmscript LANGUAGES CXX ASM_MASM)
include_directories(include)
link_directories(lib)
add_definitions(-DLOGLEVEL=4)
file(GLOB_RECURSE src CONFIGURE_DEPENDS "*.h" "*.cpp")
add_library(tmscript SHARED ${src} SuitHook.asm)
target_link_libraries(tmscript libMinHook.x64 modsettings)
set_target_properties(tmscript PROPERTIES OUTPUT_NAME "tmscript" PREFIX "")
